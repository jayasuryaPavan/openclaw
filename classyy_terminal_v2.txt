    function generateTimeSlots() {
        const container = document.getElementById('editTimeOptions');

            for (let min = 0; min < 60; min += 30) {
                if (hour === 22 && min > 0) break; // Stop at 10 PM
                const time24 = `${hour.toString().padStart(2, '0')}:${min.toString().padStart(2, '0')}`;
                const hour12 = hour > 12 ? hour - 12 : (hour === 0 ? 12 : hour);
                const ampm = hour >= 12 ? 'PM' : 'AM';
                const timeDisplay = `${hour12}:${min.toString().padStart(2, '0')} ${ampm}`;
                
                const option = document.createElement('div');
                option.className = 'custom-dropdown-option';
                option.dataset.value = time24;
                option.textContent = timeDisplay;
                option.onclick = () => selectEditTime(time24, timeDisplay);
                container.appendChild(option);
            }
        }

    function renderEditCalendar() {
        const container = document.getElementById('editCalendarDays');
        const monthYearEl = document.getElementById('editCalendarMonthYear');
        
        const months = ['January', 'February', 'March', 'April', 'May', 'June', 
                        'July', 'August', 'September', 'October', 'November', 'December'];
        
        monthYearEl.textContent = `${months[editCalendarCurrentMonth]} ${editCalendarCurrentYear}`;
        
        const firstDay = new Date(editCalendarCurrentYear, editCalendarCurrentMonth, 1).getDay();
        const daysInMonth = new Date(editCalendarCurrentYear, editCalendarCurrentMonth + 1, 0).getDate();
        const daysInPrevMonth = new Date(editCalendarCurrentYear, editCalendarCurrentMonth, 0).getDate();
        
        const today = new Date();
        const todayStr = `${today.getFullYear()}-${(today.getMonth() + 1).toString().padStart(2, '0')}-${today.getDate().toString().padStart(2, '0')}`;
        
        container.innerHTML = '';
        
        // Previous month days
        for (let i = firstDay - 1; i >= 0; i--) {
            const day = daysInPrevMonth - i;
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.className = 'p-2 text-center rounded text-gray-600 text-sm';
            btn.textContent = day;
            btn.disabled = true;
            container.appendChild(btn);
        }
        
        // Current month days
        for (let day = 1; day <= daysInMonth; day++) {
            const dateStr = `${editCalendarCurrentYear}-${(editCalendarCurrentMonth + 1).toString().padStart(2, '0')}-${day.toString().padStart(2, '0')}`;
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.className = 'p-2 text-center rounded text-sm transition-colors ';
            
            if (dateStr === todayStr) {
                btn.className += 'ring-1 ring-indigo-500 text-indigo-400 ';
            }
            
            if (editSelectedDate === dateStr) {
                btn.className += 'bg-indigo-600 text-white ';
            } else {
                btn.className += 'text-gray-300 hover:bg-gray-800 ';
            }
            
            btn.textContent = day;
            btn.onclick = () => selectEditDate(dateStr, day);
            container.appendChild(btn);
        }
        
        // Next month days
        const totalCells = Math.ceil((firstDay + daysInMonth) / 7) * 7;
        const remainingCells = totalCells - (firstDay + daysInMonth);
        for (let day = 1; day <= remainingCells; day++) {
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.className = 'p-2 text-center rounded text-gray-600 text-sm';
            btn.textContent = day;
            btn.disabled = true;
            container.appendChild(btn);
        }
    }
